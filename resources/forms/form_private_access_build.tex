\newcommand{\staticbadge}[4]{%
  \begin{scope}
    % Outer border / cutting guideline
    \draw[bordergray, line width=1.5px] (0, 0) rectangle (675px, 1012.5px);

    % Top Access Level Header Block
    \fill[#1] (0, 795px) rectangle (675px, 1012.5px);
    \fill[accentgold] (0, 786px) rectangle (675px, 795px);

    % Access Level Static Text (Uses \HeaderFontSize)
    \node[anchor=center, text=navytext] at (337.5px, 903px) {%
      \fontsize{\HeaderFontSize pt}{\dimexpr\HeaderFontSize pt*12/10\relax}\selectfont\bfseries #2%
    };

    % Photo Area / Embedded Image
    \node[anchor=center] at (337.5px, 510px) {%
      \includegraphics[width=480px, height=450px, keepaspectratio=false]{#3}%
    };

    % Central Photo Frame Border
    \draw[accentgold, line width=4.5px] (97.5px, 285px) rectangle (577.5px, 735px);

    % Name / User Field Static Text (Uses \NameFontSize)
    \node[anchor=center, text=navytext] at (337.5px, 216px) {%
      \fontsize{\NameFontSize pt}{\dimexpr\NameFontSize pt*12/10\relax}\selectfont\bfseries #4%
    };

    % Bottom Expiration Date Footer Block
    \fill[#1] (0, 0) rectangle (675px, 135px);

    % Expiration Date Static Text (Uses \ExpFontSize)
    \node[anchor=east, text=navytext] at (645px, 67.5px) {%
      \fontsize{\ExpFontSize pt}{\dimexpr\ExpFontSize pt*12/10\relax}\selectfont Exp: \ExpireDate%
    };
  \end{scope}%
}
