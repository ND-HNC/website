% Format free response
\newcommand{\fillableline}[3]{%
  \begingroup
  \large
  #1 \underline{\TextField[%
    name=#2, %
    width=#3, %
    backgroundcolor={1 1 1},
    bordercolor={1 1 1}, % White removes the native interactive box outline
    charsize=12pt
  ]{}}%
  \endgroup
  \vspace{6pt}%
}

    \stepcounter{choiceboxcount}% Increments the counter for a unique group name
    \begingroup
    \large
    \begin{Form}
        \ChoiceMenu[
            radio,
            radiosymbol=\ding{52},
            name=group\thechoiceboxcount,
            backgroundcolor={1 1 1},
            bordercolor={0 0 0},
            charsize=8pt
        ]{}{\hspace{0pt}Yes ,\hspace{3pt}No }%
        \quad#1
    \end{Form}%
    \endgroup
    \vspace{6pt}%
}
